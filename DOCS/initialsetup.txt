# You already have Homebrew. Make sure Python 3 is there:
brew install python

# Create a project folder + virtual environment (keeps things clean)
mkdir selenium-practice && cd selenium-practice
python3 -m venv venv
source venv/bin/activate

# Install Selenium
pip install selenium

Every time you come back to work on this, just cd selenium-practice and run source venv/bin/activate again. When you're done, type deactivate.
